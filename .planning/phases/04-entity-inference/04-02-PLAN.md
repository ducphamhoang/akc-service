class TestExtractEntityFromContext:

    def test_returns_direct_entity_field(self):
        """Direct entity key in akc_context is returned."""
        from akc_service.api.routes import _extract_entity_from_context
        result = _extract_entity_from_context({"entity": "physics"})

        """Falls back to knowledge_patterns_active[0].entity."""
        from akc_service.api.routes import _extract_entity_from_context

            "knowledge_patterns_active": [{"entity": "animation", "id": "p1"}]

        assert result == "animation"

    def test_direct_entity_takes_precedence(self):
        """Direct entity key wins over pattern entity."""
        from akc_service.api.routes import _extract_entity_from_context
        result = _extract_entity_from_context({
            "entity": "physics",
            "knowledge_patterns_active": [{"entity": "animation"}]
        })
        assert result == "physics"

    def test_returns_none_for_empty_context(self):
        from akc_service.api.routes import _extract_entity_from_context
        assert _extract_entity_from_context({}) is None

    def test_returns_none_for_none_context(self):
        from akc_service.api.routes import _extract_entity_from_context
        assert _extract_entity_from_context(None) is None

    def test_returns_none_when_no_entity_anywhere(self):
        from akc_service.api.routes import _extract_entity_from_context
        result = _extract_entity_from_context({
            "knowledge_patterns_active": [{"id": "p1", "component": "col"}]
